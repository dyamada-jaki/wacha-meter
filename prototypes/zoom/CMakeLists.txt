cmake_minimum_required(VERSION 3.20)
project(wacha_zoom_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SDK_ROOT "" CACHE PATH "Git 管理外の Zoom Linux ARM64 SDK")
if(NOT EXISTS "${SDK_ROOT}/libmeetingsdk.so" OR NOT EXISTS "${SDK_ROOT}/h/zoom_sdk.h")
  message(FATAL_ERROR "SDK_ROOT に SDK を指定してください（SDK はコピーしません）")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
add_executable(zoom-probe main.cpp)
target_include_directories(zoom-probe SYSTEM PRIVATE "${SDK_ROOT}/h")
target_compile_options(zoom-probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(zoom-probe PRIVATE "${SDK_ROOT}/libmeetingsdk.so" PkgConfig::GLIB)
set_target_properties(zoom-probe PROPERTIES BUILD_RPATH "${SDK_ROOT};${SDK_ROOT}/qt_libs;${SDK_ROOT}/qt_libs/Qt/lib")
